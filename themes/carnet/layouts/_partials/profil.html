{{- /* Profil altimétrique en SVG, calculé au build. Contexte : résultat de gpx-stats. */ -}} {{- $W := 1000.0 -}}{{- $H := 200.0 -}} {{- $range := sub .max .min -}} {{- $step := cond (le $range 250) 50 (cond (le $range 700) 100 200) -}} {{- $lo := mul (math.Floor (div .min $step)) $step -}} {{- $hi := mul (math.Ceil (div .max $step)) $step -}} {{- if eq $hi $lo }}{{ $hi = add $lo $step }}{{ end -}} {{- $total := math.Max .distance 1 -}} {{- /* une courbe par trace GPX, placée sur l'axe des distances de la journée */ -}} {{- $courbes := slice -}} {{- range .segments -}} {{- $seg := . -}} {{- $line := "" -}}{{- $x0 := 0.0 -}}{{- $x1 := 0.0 -}} {{- range $i, $e := $seg.ele -}} {{- $x := mul (div (index $seg.d $i) $total) $W -}} {{- $y := sub $H (mul (div (sub $e $lo) (sub $hi $lo)) $H) -}} {{- if eq $i 0 }}{{ $x0 = $x }}{{ end }}{{ $x1 = $x -}} {{- $line = printf "%s%.1f,%.1f " $line $x $y -}} {{- end -}} {{- $courbes = $courbes | append (dict "ligne" $line "x0" $x0 "x1" $x1) -}} {{- end -}}
