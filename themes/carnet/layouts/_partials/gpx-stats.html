{{- /* Traces GPX d'une étape, combinées. Front matter : gpx: "trace.gpx" # un fichier gpx: ["matin.gpx", "aprem.gpx"] # plusieurs, dans l'ordre de la journée Sans « gpx » : tous les .gpx du dossier, par ordre de nom. Chaque fichier est calculé séparément (gpx-trace.html) puis les chiffres sont additionnés : le trajet entre deux traces (bus, voiture…) n'est pas compté dans la distance ni dans le dénivelé. Usage : {{ $s := partialCached "gpx-stats.html" . .RelPermalink }} Renvoie false si l'étape n'a pas de GPX, sinon : distance, up, down, min, max, fichiers (ressources), segments (une entrée par fichier : lat, lon, ele, d en distance cumulée sur la journée) */ -}} {{- $noms := .Params.gpx -}} {{- if and $noms (not (reflect.IsSlice $noms)) }}{{ $noms = slice $noms }}{{ end -}} {{- $fichiers := slice -}} {{- if $noms -}} {{- range $noms -}} {{- with $.Resources.Get . -}} {{- $fichiers = $fichiers | append . -}} {{- else -}} {{- warnf "%s : fichier GPX « %s » introuvable dans le dossier de l'étape" $.File.Path . -}} {{- end -}} {{- end -}} {{- else -}} {{- range sort (.Resources.Match "*.gpx") "Name" }}{{ $fichiers = $fichiers | append . }}{{ end -}} {{- end -}} {{- $segments := slice -}}{{- $retenus := slice -}} {{- $dist := 0.0 -}}{{- $up := 0.0 -}}{{- $down := 0.0 -}} {{- $min := 100000.0 -}}{{- $max := -100000.0 -}} {{- range $fichiers -}} {{- with partialCached "gpx-trace.html" . .RelPermalink -}} {{- $decale := slice -}} {{- range .d }}{{ $decale = $decale | append (add . (math.Round $dist)) }}{{ end -}} {{- $segments = $segments | append (dict "lat" .lat "lon" .lon "ele" .ele "d" $decale) -}} {{- $retenus = $retenus | append .fichier -}} {{- $dist = add $dist .distance -}}{{- $up = add $up .up -}}{{- $down = add $down .down -}} {{- $min = math.Min $min .min -}}{{- $max = math.Max $max .max -}} {{- end -}} {{- end -}} {{- $result := false -}} {{- with $segments -}} {{- $result = dict "distance" $dist "up" $up "down" $down "min" $min "max" $max "fichiers" $retenus "segments" $segments -}} {{- end -}} {{- return $result -}}
