{{- /* Lit le fichier GPX de l'étape (param « gpx » du front matter, sinon le premier .gpx du dossier) et calcule, au build : distance (m), D+, D- (m), altitudes min/max et les tableaux de points. Usage : {{ $s := partialCached "gpx-stats.html" . .RelPermalink }} Renvoie false si l'étape n'a pas de GPX. */ -}} {{- $result := false -}} {{- $fichier := false -}} {{- with .Params.gpx -}} {{- $fichier = $.Resources.Get . -}} {{- if not $fichier }}{{ warnf "%s : fichier GPX « %s » introuvable dans le dossier de l'étape" $.File.Path . }}{{ end -}} {{- else -}} {{- $fichier = .Resources.GetMatch "*.gpx" -}} {{- end -}} {{- with $fichier -}} {{- $gpx := transform.Unmarshal (dict "format" "xml") .Content -}} {{- $seuil := site.Params.seuilDenivele | default 3 -}} {{- $R := 6371000.0 -}} {{- $lats := slice -}}{{- $lons := slice -}}{{- $eles := slice -}}{{- $ds := slice -}} {{- $dist := 0.0 -}}{{- $up := 0.0 -}}{{- $down := 0.0 -}} {{- $min := 100000.0 -}}{{- $max := -100000.0 -}} {{- $plat := 0.0 -}}{{- $plon := 0.0 -}}{{- $ref := 0.0 -}}{{- $first := true -}} {{- /* trk / trkseg / trkpt sont des objets s'il n'y en a qu'un, des listes sinon */ -}} {{- $trks := $gpx.trk -}} {{- if not (reflect.IsSlice $trks) }}{{ $trks = slice $trks }}{{ end -}} {{- range $trks -}} {{- $segs := .trkseg -}} {{- if not (reflect.IsSlice $segs) }}{{ $segs = slice $segs }}{{ end -}} {{- range $segs -}} {{- $pts := .trkpt -}} {{- if not (reflect.IsSlice $pts) }}{{ $pts = slice $pts }}{{ end -}} {{- range $pts -}} {{- $lat := float (index . "-lat") -}} {{- $lon := float (index . "-lon") -}} {{- $ele := float (.ele | default $ref) -}} {{- if $first -}} {{- $ref = $ele -}}{{- $first = false -}} {{- else -}} {{- /* distance haversine depuis le point précédent */ -}} {{- $p1 := math.ToRadians $plat -}}{{- $p2 := math.ToRadians $lat -}} {{- $dp := math.ToRadians (sub $lat $plat) -}} {{- $dl := math.ToRadians (sub $lon $plon) -}} {{- $a := add (math.Pow (math.Sin (div $dp 2)) 2) (mul (math.Cos $p1) (math.Cos $p2) (math.Pow (math.Sin (div $dl 2)) 2)) -}} {{- $dist = add $dist (mul 2 $R (math.Atan2 (math.Sqrt $a) (math.Sqrt (sub 1 $a)))) -}} {{- /* dénivelé avec hystérésis pour filtrer le bruit */ -}} {{- $diff := sub $ele $ref -}} {{- if ge $diff $seuil -}} {{- $up = add $up $diff -}}{{- $ref = $ele -}} {{- else if le $diff (mul -1 $seuil) -}} {{- $down = sub $down $diff -}}{{- $ref = $ele -}} {{- end -}} {{- end -}} {{- $min = math.Min $min $ele -}}{{- $max = math.Max $max $ele -}} {{- $plat = $lat -}}{{- $plon = $lon -}} {{- $lats = $lats | append (div (math.Round (mul $lat 100000)) 100000) -}} {{- $lons = $lons | append (div (math.Round (mul $lon 100000)) 100000) -}} {{- $eles = $eles | append (div (math.Round (mul $ele 10)) 10) -}} {{- $ds = $ds | append (math.Round $dist) -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $lats) 1 -}} {{- $result = dict "gpx" . "distance" $dist "up" $up "down" $down "min" $min "max" $max "lat" $lats "lon" $lons "ele" $eles "d" $ds -}} {{- end -}} {{- end -}} {{- return $result -}}
