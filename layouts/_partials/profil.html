{{- /* Profil altimétrique en SVG, calculé au build. Contexte : résultat de gpx-stats. */ -}} {{- $W := 1000.0 -}}{{- $H := 200.0 -}} {{- $range := sub .max .min -}} {{- $step := cond (le $range 250) 50 (cond (le $range 700) 100 200) -}} {{- $lo := mul (math.Floor (div .min $step)) $step -}} {{- $hi := mul (math.Ceil (div .max $step)) $step -}} {{- if eq $hi $lo }}{{ $hi = add $lo $step }}{{ end -}} {{- $total := math.Max .distance 1 -}} {{- $line := "" -}} {{- range $i, $e := .ele -}} {{- $x := mul (div (index $.d $i) $total) $W -}} {{- $y := sub $H (mul (div (sub $e $lo) (sub $hi $lo)) $H) -}} {{- $line = printf "%s%.1f,%.1f " $line $x $y -}} {{- end -}}
